🔍 2. ระบบคัดกรองหมวดหมู่และค้นหาเรียลไทม์ (Filter & Search)

ทำหน้าที่กรองการ์ดสถานที่ท่องเที่ยวโดยอัตโนมัติเมื่อผู้ใช้คลิกเลือกปุ่มหมวดหมู่ หรือพิมพ์คำค้นหาในช่อง Search Box

function filterCards() JavaS-1.js (Line 10-35)

วัตถุประสงค์: ตรวจสอบเงื่อนไข 2 ทางพร้อมกัน (หมวดหมู่ที่เลือก + ข้อความที่พิมพ์ค้นหา) และซ่อน/แสดงการ์ดท่องเที่ยวแบบเรียลไทม์

JavaScript Source Code
function filterCards() {
  const searchText = searchInput ? searchInput.value.toLowerCase().trim() : "";
  let visibleCount = 0;

  cards.forEach(card => {
    const category = card.dataset.category || "";
    const title = card.querySelector("h3") ? card.querySelector("h3").textContent.toLowerCase() : "";
    const location = card.querySelector(".location") ? card.querySelector(".location").textContent.toLowerCase() : "";

    const matchesCategory = (activeCategory === "ทั้งหมด" || category === activeCategory);
    const matchesSearch = !searchText || title.includes(searchText) || location.includes(searchText);

    if (matchesCategory && matchesSearch) {
      card.style.display = "block";
      visibleCount++;
    } else {
      card.style.display = "none";
    }
  });

  if (noResults) {
    noResults.style.display = (visibleCount === 0) ? "block" : "none";
  }
}

วัตถุประสงค์: ดักจับ Event พิกัดเมาส์ (e.pageX, e.pageY) แล้วย้ายตำแหน่งป๊อบอัพลอยให้เคลื่อนที่ตามเมาส์ขณะจี้การ์ด

JavaScript Event Listeners 
cards.forEach(card => {
  card.addEventListener("mouseenter", function () {
    // ดึงเฉพาะ URL รูปภาพจากการ์ดใบที่เลื่อนเมาส์ผ่าน
    const img = card.querySelector("img") ? card.querySelector("img").src : "";
    if (previewImg) previewImg.src = img;
    popup.style.display = "block";
  });

  card.addEventListener("mousemove", function (e) {
    popup.style.top = (e.pageY + 20) + "px";
    popup.style.left = (e.pageX + 20) + "px";
  });

  card.addEventListener("mouseleave", function () {
    popup.style.display = "none";
  });
});

calculateHomeTotalPrice() JavaS-1.js (Line 120-136)

วัตถุประสงค์: คำนวณยอดเงินรวม (ราคาต่อคน × จำนวนคน) แล้วฟอร์แมตให้อยู่ในรูปแบบสกุลเงินไทยบาท (เช่น ฿4,500)

JavaScript Source Code
function calculateHomeTotalPrice() {
  const countEl = document.getElementById("homeTravelerCount");
  const count = countEl ? parseInt(countEl.value) || 1 : 1;
  const total = currentHomeUnitPrice * count;

  const totalEl = document.getElementById("homeTotalPriceDisplay");
  if (totalEl) {
    totalEl.textContent = "฿" + total.toLocaleString();
  }
  return total;
}

🔬 เทคนิคสำคัญที่ใช้:

1. parseInt() แปลงข้อความจากตัวเลือก <select> ให้เป็นตัวเลขสำหรับคำนวณทางคณิตศาสตร์

🗺️ 5. ระบบแผนที่และการนำทางแบบเรียลไทม์ (Geolocation API)

ดึงตำแหน่งพิกัด GPS ปัจจุบันของผู้ใช้ผ่านเบราว์เซอร์ แล้วสร้างลิงก์คำนวณเส้นทางไปยังสถานที่ท่องเที่ยวนั้นๆ ผ่าน Google Maps Embed

function loadLiveRoute(...) หน้าย่อย เช่น เขาใหญ่-1.html

วัตถุประสงค์: เรียกใช้ Web API navigator.geolocation เพื่อนำพิกัดละติจูด/ลองจิจูดของผู้ใช้ไปปักจุดเริ่มต้นการเดินทาง

JavaScript Geolocation API 
function loadLiveRoute(destinationQuery, iframeId, statusMsgId) {
  const iframe = document.getElementById(iframeId);
  const statusMsg = document.getElementById(statusMsgId);

  if ("geolocation" in navigator) {
    statusMsg.textContent = "⏳ กำลังค้นหาตำแหน่งปัจจุบันของคุณ...";
    navigator.geolocation.getCurrentPosition(
      function(pos) {
        const lat = pos.coords.latitude;
        const lng = pos.coords.longitude;
        if (iframe) {
          iframe.src = "https://maps.google.com/maps?saddr=" + lat + "," + lng + "&daddr=" + encodeURIComponent(destinationQuery) + "&output=embed";
        }
        statusMsg.textContent = "✅ แสดงเส้นทางจากตำแหน่งของคุณสำเร็จ!";
      },
      function(err) {
        // หากผู้ใช้ไม่อนุญาต ให้ใช้ตำแหน่งคร่าวๆ
        iframe.src = "https://maps.google.com/maps?saddr=My+Location&daddr=" + encodeURIComponent(destinationQuery) + "&output=embed";
      }
    );
  }
}

วัตถุประสงค์: อ่านข้อมูล JSON อาร์เรย์ที่บันทึกไว้ใน localStorage.getItem("myBookings") แล้วสร้างการ์ดรายการประวัติการจองแบบไดนามิก

JavaScript LocalStorage Operations 
function saveBookingToStorage(bookingObj) {
  let history = JSON.parse(localStorage.getItem("travel_bookings") || "[]");
  history.unshift(bookingObj); // ใส่ข้อมูลใหม่ไว้ด้านหน้าสุด
  localStorage.setItem("travel_bookings", JSON.stringify(history));
}
